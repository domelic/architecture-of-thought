\chapter{DCF Principles Summary}

\section{Core Principles}

\begin{enumerate}
    \item \textbf{The Thinking Mirror}: LLMs reflect and transform your thought; quality input yields quality output
    \item \textbf{Socratic Dialogue}: Use questioning to arrive at clarity, not prompting to extract answers
    \item \textbf{Prompt Chaining}: Decompose complex tasks into manageable, sequential operations (now often automated)
    \item \textbf{Recursive Refinement}: Output becomes input; iterate until convergence
    \item \textbf{Language as Infrastructure}: Documentation and CLAUDE.md files are cognitive engineering
\end{enumerate}

\section{Metacognitive Principles}

\begin{enumerate}
    \setcounter{enumi}{5}
    \item \textbf{Metacognition}: Think about thinking; reflect on process, not just outcome
    \item \textbf{Anticipatory Calibration}: Form hypotheses about AI output before prompting; surprise reveals miscalibration
    \item \textbf{The Meta-Question}: ``What question should I be asking?'' is often more valuable than any answer; finding the right question unlocks understanding
    \item \textbf{Extended Mind}: LLMs are cognitive augmentation, expanding the boundaries of thought
    \item \textbf{Scaffolding}: AI support builds independent capability; the goal is growth, not dependence
\end{enumerate}

\section{Collaboration Principles}

\begin{enumerate}
    \setcounter{enumi}{10}
    \item \textbf{Co-creation}: Neither human nor AI alone; together, capabilities emerge that neither possessed
    \item \textbf{Continuous Evolution}: The methodology grows; document what works; share with others
\end{enumerate}

\section{Agentic Era Principles}

\begin{enumerate}
    \setcounter{enumi}{12}
    \item \textbf{Checkpoint Engagement}: Apply DCF at approval points, not every interaction
    \item \textbf{Tool Selection Awareness}: Evaluate whether the right tool was chosen for the task
    \item \textbf{Configuration as Philosophy}: Permission settings reflect cognitive engagement strategy
    \item \textbf{Memory Maintenance}: Actively curate CLAUDE.md and memories as cognitive infrastructure
\end{enumerate}
